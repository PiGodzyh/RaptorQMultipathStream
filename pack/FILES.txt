📚 主要 API
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

namespace RQPack {

  📤 编码器类
  class Encoder {
      Encoder(data, symbol_size)      // 构造函数
      encode(symbol_id)                // 编码单个符号
      encodeAll(repair_count)          // 批量编码
      getSourceSymbolCount()           // 获取源符号数量
      getMaxRepairSymbolCount()        // 获取最大修复符号数
  };

  📥 解码器类
  class Decoder {
      Decoder(data_size, symbol_size)  // 构造函数
      addSymbol(symbol)                // 添加符号
      addSymbols(symbols)              // 批量添加符号
      canDecode()                      // 检查是否可解码
      neededSymbols()                  // 还需要的符号数
      decode()                         // 执行解码
  };

  📦 符号结构
  struct Symbol {
      uint32_t id;                     // 符号 ID
      std::vector<uint8_t> data;       // 符号数据
  };

  🔧 工具函数
  quickEncode(data, symbol_size, repair_ratio)   // 一键编码
  quickDecode(symbols, data_size, symbol_size)   // 一键解码
}

💡 使用示例
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

#include "rq_pack.h"
using namespace RQPack;

// 编码
std::vector<uint8_t> data = {1, 2, 3, 4, 5};
auto symbols = quickEncode(data, 64, 0.1f);

// 解码
auto decoded = quickDecode(symbols, data.size(), 64);
